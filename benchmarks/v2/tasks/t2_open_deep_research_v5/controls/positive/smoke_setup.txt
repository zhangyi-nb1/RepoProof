cat > langchain/chat_models.py <<'RP_SETUP'
"""`init_chat_model` 的最小实现 —— OpenAI 兼容 Chat Runnable。

只补 ODR 图实际用到的那几个方法(with_config / bind_tools /
with_structured_output / with_retry / ainvoke)。端点取
OPENAI_BASE_URL + OPENAI_API_KEY:oracle 的 fake_llm_stub 就注在这两个
环境变量上,所以模型响应是**真的经这条线回到图里**的 —— h1 的金丝雀与
请求数两条因果判据靠的正是这一点。
"""
import asyncio
import json
import os
import urllib.request
from copy import copy

from langchain_core.messages import AIMessage
from langchain_core.utils.function_calling import convert_to_openai_tool


def _message(m):
    if isinstance(m, dict):
        return {"role": m.get("role", "user"), "content": m.get("content", "")}
    role = {"human": "user", "ai": "assistant", "system": "system",
            "tool": "tool"}.get(getattr(m, "type", "human"), "user")
    out = {"role": role, "content": m.content or ""}
    if role == "tool":
        out["tool_call_id"] = getattr(m, "tool_call_id", "call_0")
    return out


class _Chat:
    def __init__(self, options=None, tools=None, schema=None):
        self.options = options or {}
        self.tools = tools
        self.schema = schema

    def _clone(self, **kw):
        x = copy(self)
        for k, v in kw.items():
            setattr(x, k, v)
        return x

    def with_config(self, config=None, **kw):
        opts = dict(self.options)
        opts.update(config or {})
        opts.update(kw)
        return self._clone(options=opts)

    def with_retry(self, **kw):
        return self

    def bind_tools(self, tools, **kw):
        return self._clone(tools=list(tools))

    def with_structured_output(self, schema, **kw):
        return self._clone(schema=schema)

    def invoke(self, messages, config=None, **kw):
        return self._invoke(messages, config or {})

    async def ainvoke(self, messages, config=None, **kw):
        return await asyncio.to_thread(self._invoke, messages, config or {})

    def _invoke(self, messages, runtime):
        opts = dict(self.options)
        opts.update((runtime or {}).get("configurable", {}))
        base = os.environ.get("OPENAI_BASE_URL", "https://api.openai.com/v1").rstrip("/")
        key = opts.get("api_key") or os.environ.get("OPENAI_API_KEY", "")
        model = str(opts.get("model") or "openai:gpt-4.1").split(":", 1)[-1]
        body = {"model": model,
                "messages": [_message(m) for m in messages],
                "max_tokens": int(opts.get("max_tokens") or 4096)}
        tools = self.tools
        if self.schema is not None:
            tools = [self.schema]
        if tools:
            body["tools"] = [convert_to_openai_tool(t) for t in tools]
            body["tool_choice"] = "auto"
        req = urllib.request.Request(
            base + "/chat/completions",
            data=json.dumps(body).encode(),
            headers={"Content-Type": "application/json",
                     "Authorization": f"Bearer {key}"})
        with urllib.request.urlopen(req, timeout=60) as r:
            data = json.loads(r.read().decode())
        msg = (data.get("choices") or [{}])[0].get("message") or {}
        text = msg.get("content") or ""
        calls = msg.get("tool_calls") or []
        if self.schema is not None:
            args = {}
            if calls:
                try:
                    args = json.loads(calls[0]["function"]["arguments"] or "{}")
                except Exception:
                    args = {}
            if not args:
                try:
                    args = json.loads(text)
                except Exception:
                    args = {}
            fields = getattr(self.schema, "model_fields", {})
            for name, f in fields.items():
                if name not in args:
                    ann = getattr(f, "annotation", str)
                    args[name] = [] if ann in (list,) else (text if ann is str else None)
            return self.schema(**args)
        tool_calls = []
        for i, c in enumerate(calls):
            fn = c.get("function") or {}
            try:
                a = json.loads(fn.get("arguments") or "{}")
            except Exception:
                a = {}
            tool_calls.append({"name": fn.get("name", ""), "args": a,
                               "id": c.get("id") or f"call_{i}", "type": "tool_call"})
        return AIMessage(content=text, tool_calls=tool_calls)


def init_chat_model(model=None, *, configurable_fields=None, max_tokens=None,
                    api_key=None, tags=None, **kw):
    opts = {}
    if model is not None:
        opts["model"] = model
    if max_tokens is not None:
        opts["max_tokens"] = max_tokens
    if api_key is not None:
        opts["api_key"] = api_key
    return _Chat(options=opts)
